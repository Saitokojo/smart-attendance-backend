package com.bbu.attendance.security;

import com.bbu.attendance.entity.User;
import com.bbu.attendance.repository.UserRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.security.core.userdetails.UserDetailsService;
import org.springframework.security.core.userdetails.UsernameNotFoundException;
import org.springframework.stereotype.Service;

@Service
@RequiredArgsConstructor
public class CustomUserDetailsService implements UserDetailsService {

    private final UserRepository userRepository;

    @Override
    public UserDetails loadUserByUsername(String email) throws UsernameNotFoundException {
        // Fetch the user from the database
        User user = userRepository.findByEmail(email)
                .orElseThrow(() -> new UsernameNotFoundException("User not found with email : " + email));

        // Convert the user's roles into Spring Security SimpleGrantedAuthority objects
        var authorities = java.util.Collections.singletonList(
                new SimpleGrantedAuthority(user.getRole().getName())
        );
        // Check if the user's status is ACTIVE
        boolean isActive = "ACTIVE".equalsIgnoreCase(user.getStatus());
        // Return a Spring Security User object
        return new org.springframework.security.core.userdetails.User(
                user.getEmail(),          // Using email as the principal
                user.getPasswordHash(),   // Matches your passwordHash field
                isActive,                 // Checked against your String status field
                true,                     // account non-expired
                true,                     // credentials non-expired
                true,                     // account non-locked
                authorities
        );
    }
}

package com.bbu.attendance.service.impl;

import com.bbu.attendance.dto.RoleRequest;
import com.bbu.attendance.dto.RoleResponse;
import com.bbu.attendance.entity.Role;
import com.bbu.attendance.repository.RoleRepository;
import com.bbu.attendance.service.RoleService;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;

import java.util.List;

@Service
@RequiredArgsConstructor
public class RoleServiceImpl implements RoleService {

    private final RoleRepository roleRepository;

    @Override
    public RoleResponse createRole(RoleRequest request) {
        if (roleRepository.existsByName(request.name())) {
            throw new RuntimeException("Role name already exists");
        }

        Role role = Role.builder()
                .name(request.name())
                .description(request.description())
                .build();

        Role savedRole = roleRepository.save(role);
        return mapToResponse(savedRole);
    }

    @Override
    public List<RoleResponse> getAllRoles() {
        return roleRepository.findAll().stream()
                .map(this::mapToResponse)
                .toList();
    }

    private RoleResponse mapToResponse(Role role) {
        return new RoleResponse(
            role.getId(),
            role.getName(),
            role.getDescription()
        );
    }
}

==== UserServiceImpl.java ====
package com.bbu.attendance.service.impl;

import com.bbu.attendance.dto.UserRequest;
import com.bbu.attendance.dto.UserResponse;
import com.bbu.attendance.entity.Role;
import com.bbu.attendance.entity.User;
import com.bbu.attendance.repository.RoleRepository;
import com.bbu.attendance.repository.UserRepository;
import com.bbu.attendance.service.UserService;
import lombok.RequiredArgsConstructor;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;

@Service
@RequiredArgsConstructor
public class UserServiceImpl implements UserService {

    private final UserRepository userRepository;
    private final RoleRepository roleRepository;
    private final PasswordEncoder passwordEncoder; // Requires a PasswordEncoder bean in your SecurityConfig

    @Override
    public UserResponse createUser(UserRequest request) {
        if (userRepository.existsByEmail(request.email())) {
            throw new RuntimeException("Email already in use");
        }

        Role role = roleRepository.findById(request.roleId())
                .orElseThrow(() -> new RuntimeException("Role not found"));

        User user = User.builder()
                .name(request.name())
                .email(request.email())
                .phone(request.phone())
                .studentCode(request.studentCode())
                .employeeCode(request.employeeCode())
                .passwordHash(passwordEncoder.encode(request.password())) // Hash the password
                .role(role)
                .status("ACTIVE")
                .build();

        User savedUser = userRepository.save(user);
        return mapToResponse(savedUser);
    }

    @Override
    public UserResponse getUserByEmail(String email) {
        User user = userRepository.findByEmail(email)
                .orElseThrow(() -> new RuntimeException("User not found"));
        return mapToResponse(user);
    }

    // Helper method to convert Entity to DTO
    private UserResponse mapToResponse(User user) {
        return new UserResponse(
                user.getId(),
                user.getName(),
                user.getEmail(),
                user.getPhone(),
                user.getStudentCode(),
                user.getEmployeeCode(),
                user.getRole().getName(),
                user.getStatus()
        );
    }
}

-- ====================================================================

-- 1. ROLES TABLE
CREATE TABLE IF NOT EXISTS roles (
    id SERIAL PRIMARY KEY,
    name VARCHAR(50) UNIQUE NOT NULL, -- 'STUDENT', 'TEACHER', 'ADMIN'
    description TEXT
);

-- 2. USERS TABLE
CREATE TABLE IF NOT EXISTS users (
    id BIGSERIAL PRIMARY KEY,
    role_id INT NOT NULL REFERENCES roles(id),
    student_code VARCHAR(50) UNIQUE,
    employee_code VARCHAR(50) UNIQUE,
    name VARCHAR(150) NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    phone VARCHAR(30),
    password_hash VARCHAR(255) NOT NULL,
    status VARCHAR(30) DEFAULT 'ACTIVE',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 3. DEVICES TABLE (Device Binding)
CREATE TABLE IF NOT EXISTS devices (
    id BIGSERIAL PRIMARY KEY,
    user_id BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    device_uuid VARCHAR(255) UNIQUE NOT NULL,
    device_name VARCHAR(100),
    platform VARCHAR(50),
    is_active BOOLEAN DEFAULT TRUE,
    registered_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    last_used_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 4. ROOMS TABLE (Geofence & Multi-floor)
CREATE TABLE IF NOT EXISTS rooms (
    id SERIAL PRIMARY KEY,
    room_code VARCHAR(50) UNIQUE NOT NULL,
    building_name VARCHAR(100) NOT NULL,
    floor_number INT NOT NULL DEFAULT 1,
    latitude DECIMAL(10, 8) NOT NULL,
    longitude DECIMAL(11, 8) NOT NULL,
    geofence_radius DECIMAL(5, 2) DEFAULT 40.00,
    status VARCHAR(30) DEFAULT 'ACTIVE'
);

-- 5. CLASSES TABLE (with Invite Code)
CREATE TABLE IF NOT EXISTS classes (
    id BIGSERIAL PRIMARY KEY,
    code VARCHAR(50) UNIQUE NOT NULL,
    name VARCHAR(200) NOT NULL,
    description TEXT,
    teacher_id BIGINT NOT NULL REFERENCES users(id),
    invite_code VARCHAR(10) UNIQUE NOT NULL,
    is_invite_open BOOLEAN DEFAULT TRUE,
    invite_expires_at TIMESTAMP WITH TIME ZONE NULL,
    max_students INT DEFAULT 60,
    status VARCHAR(30) DEFAULT 'ACTIVE',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 6. CLASS_STUDENTS TABLE (Enrollment)
CREATE TABLE IF NOT EXISTS class_students (
    id BIGSERIAL PRIMARY KEY,
    class_id BIGINT NOT NULL REFERENCES classes(id) ON DELETE CASCADE,
    student_id BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    status VARCHAR(30) DEFAULT 'ENROLLED',
    enrolled_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT unique_student_class UNIQUE(class_id, student_id)
);

-- 7. CLASS_SCHEDULES TABLE
CREATE TABLE IF NOT EXISTS class_schedules (
    id BIGSERIAL PRIMARY KEY,
    class_id BIGINT NOT NULL REFERENCES classes(id) ON DELETE CASCADE,
    room_id INT NOT NULL REFERENCES rooms(id),
    day_of_week INT NOT NULL, -- 1: Monday, 7: Sunday
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    status VARCHAR(30) DEFAULT 'ACTIVE'
);

-- 8. ATTENDANCE_SESSIONS TABLE
CREATE TABLE IF NOT EXISTS attendance_sessions (
    id BIGSERIAL PRIMARY KEY,
    class_id BIGINT NOT NULL REFERENCES classes(id),
    teacher_id BIGINT NOT NULL REFERENCES users(id),
    room_id INT NOT NULL REFERENCES rooms(id),
    schedule_id BIGINT REFERENCES class_schedules(id),
    session_date DATE NOT NULL,
    start_time TIMESTAMP WITH TIME ZONE NOT NULL,
    end_time TIMESTAMP WITH TIME ZONE,
    late_threshold_minutes INT DEFAULT 15,
    qr_secret_salt VARCHAR(100) NOT NULL,
    status VARCHAR(30) DEFAULT 'ACTIVE',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    closed_at TIMESTAMP WITH TIME ZONE
);

-- 9. ATTENDANCE_RECORDS TABLE (7-Factor Verification)
CREATE TABLE IF NOT EXISTS attendance_records (
    id BIGSERIAL PRIMARY KEY,
    session_id BIGINT NOT NULL REFERENCES attendance_sessions(id),
    student_id BIGINT NOT NULL REFERENCES users(id),
    status VARCHAR(30) NOT NULL, -- 'PRESENT', 'LATE', 'ABSENT', 'EXCUSED', 'N/A'
    check_in_time TIMESTAMP WITH TIME ZONE,
    client_latitude DECIMAL(10, 8),
    client_longitude DECIMAL(11, 8),
    client_ip VARCHAR(50),
    device_id BIGINT REFERENCES devices(id),
    biometric_verified BOOLEAN DEFAULT FALSE,
    wifi_ip_verified BOOLEAN DEFAULT FALSE,
    location_verified BOOLEAN DEFAULT FALSE,
    qr_verified BOOLEAN DEFAULT FALSE,
    is_manual_override BOOLEAN DEFAULT FALSE,
    override_by BIGINT REFERENCES users(id),
    override_reason TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT unique_session_student UNIQUE(session_id, student_id)
);

-- 10. LEAVE_REQUESTS TABLE
CREATE TABLE IF NOT EXISTS leave_requests (
    id BIGSERIAL PRIMARY KEY,
    student_id BIGINT NOT NULL REFERENCES users(id),
    class_id BIGINT NOT NULL REFERENCES classes(id),
    session_id BIGINT REFERENCES attendance_sessions(id),
    date_from DATE NOT NULL,
    date_to DATE NOT NULL,
    reason VARCHAR(100) NOT NULL,
    description TEXT,
    attachment_path VARCHAR(255),
    status VARCHAR(30) DEFAULT 'PENDING',
    reviewed_by BIGINT REFERENCES users(id),
    reviewed_at TIMESTAMP WITH TIME ZONE,
    review_comment TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 11. TEACHER_ABSENCE_REPORTS TABLE
CREATE TABLE IF NOT EXISTS teacher_absence_reports (
    id BIGSERIAL PRIMARY KEY,
    student_id BIGINT NOT NULL REFERENCES users(id),
    session_id BIGINT NOT NULL REFERENCES attendance_sessions(id),
    reported_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    status VARCHAR(30) DEFAULT 'PENDING',
    reviewed_by BIGINT REFERENCES users(id),

CREATE TABLE IF NOT EXISTS attendance_disputes (
